
import cv2

camera = cv2.VideoCapture(0, cv2.CAP_DSHOW)

if not camera.isOpened():
    print("Camera not detected!")
    exit()

def show_format():
    fourcc = int(camera.get(cv2.CAP_PROP_FOURCC))
    codec = "".join(chr((fourcc >> (8 * i)) & 255) for i in range(4))
    print("Current format:", codec)

print("Before changing:")
show_format()

# Try to switch to MJPG
success = camera.set(
    cv2.CAP_PROP_FOURCC,
    cv2.VideoWriter_fourcc(*"MJPG")
)

print("MJPG setting accepted:", success)

print("After changing:")
show_format()

camera.set(cv2.CAP_PROP_FRAME_WIDTH, 1280)
camera.set(cv2.CAP_PROP_FRAME_HEIGHT, 720)
camera.set(cv2.CAP_PROP_FPS, 60)

print("After resolution and FPS settings:")
show_format()

print("Reported FPS:", camera.get(cv2.CAP_PROP_FPS))

camera.release()
