
import cv2
import time

camera = cv2.VideoCapture(0, cv2.CAP_DSHOW)

if not camera.isOpened():
    print("Camera not detected!")
    exit()

camera.set(cv2.CAP_PROP_FOURCC, cv2.VideoWriter_fourcc(*"MJPG"))
camera.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
camera.set(cv2.CAP_PROP_FRAME_HEIGHT, 480)
camera.set(cv2.CAP_PROP_FPS, 60)

print("Requested FPS: 60")
print("Reported FPS:", camera.get(cv2.CAP_PROP_FPS))
print("Resolution:",
      camera.get(cv2.CAP_PROP_FRAME_WIDTH),
      "x",
      camera.get(cv2.CAP_PROP_FRAME_HEIGHT))

fourcc = int(camera.get(cv2.CAP_PROP_FOURCC))
codec = "".join(chr((fourcc >> (8 * i)) & 255) for i in range(4))
print("Video format:", codec)

# Warm up the camera
for i in range(15):
    camera.read()

# Measure actual FPS
frames = 0
start = time.perf_counter()

while frames < 120:
    ret, frame = camera.read()
    if not ret:
        break
    frames += 1

elapsed = time.perf_counter() - start
print("Measured FPS:", round(frames / elapsed, 2))

camera.release()
