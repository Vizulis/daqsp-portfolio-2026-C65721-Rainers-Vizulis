
def save_images(data, trigger_time=None):
    if trigger_time is None:
        trigger_time = datetime.now()

    image_save_dir = os.path.join(
        SAVE_DIR,
        f"images_{trigger_time.strftime('%Y%m%d_%H%M%S')}"
    )

    os.makedirs(image_save_dir, exist_ok=True)

    for i, (timestamp, frame) in enumerate(data):
        filename = os.path.join(
            image_save_dir,
            f"frame_{i:04d}_{timestamp:.6f}.png"
        )
        cv2.imwrite(filename, frame)

    print(f"Images saved in {image_save_dir}")
