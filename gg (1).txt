
def wait_for_hw_trigger():
    """Wait for a falling edge on PFI0."""
    with nidaqmx.Task() as task:
        task.di_channels.add_di_chan(TRIGGER_CHANNEL)

        previous_state = task.read()
        start_time = time.time()

        while time.time() - start_time < DAQ_TIMEOUT:
            if stop_flag.is_set():
                return False

            current_state = task.read()

            if previous_state and not current_state:
                return True

            previous_state = current_state
            time.sleep(0.001)

    return False
